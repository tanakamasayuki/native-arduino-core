compiler.c.cmd=gcc
compiler.cpp.cmd=g++
compiler.ar.cmd=ar
compiler.objcopy.cmd=objcopy
compiler.size.cmd=size
compiler.ar.flags=rcs
compiler.ar.extra_flags=

# build.extra_flags
build.extra_flags={build.extra_flags_sdl2} {build.extra_flags_m5gfx_board} {build.extra_flags_m5gfx_scale} {build.extra_flags_m5gfx_rotation}

# (No upload.tool.default) Upload should be handled by recipe.upload.pattern; avoid tools.* usage in boards.

# Compiler and linker flags
compiler.c.flags=-std=gnu11 -O2 -Wall -ffunction-sections -fdata-sections
compiler.cxx.flags=-std=gnu++11 -O2 -Wall -ffunction-sections -fdata-sections
compiler.ldflags=-Wl,--gc-sections
# macOS uses Apple's linker (ld64), which does not understand the GNU option
# "--gc-sections"; its equivalent is "-dead_strip".
compiler.ldflags.macosx=-Wl,-dead_strip

# Compile C source to object (add include paths for core and sketch)
recipe.c.o.pattern={compiler.path}{compiler.c.cmd} {compiler.c.flags} {build.extra_flags} {includes} -I{runtime.platform.path}/cores/{build.core} -I{build.path}/sketch -c -o {object_file} {source_file}

# Compile C++ source to object
recipe.cpp.o.pattern={compiler.path}{compiler.cpp.cmd} {compiler.cxx.flags} {build.extra_flags} {includes} -I{runtime.platform.path}/cores/{build.core} -I{build.path}/sketch -c -o {object_file} {source_file}

# Preprocessor recipe for library detection and function-prototype generation.
# Force "-x c++" and "-E" so the merged sketch is always treated as C++ and
# preprocessed. Some arduino-cli versions feed this step a file named
# "<sketch>.ino.cpp.merged"; without "-x c++" g++ does not recognize the
# ".merged" extension and treats it as a linker input ("linker input file
# unused"), producing no output. That makes prototype generation fail on
# Windows with "open ...\\sketch_merged.cpp: The system cannot find the file
# specified." Defining this recipe makes the behavior consistent across
# platforms and arduino-cli versions.
recipe.preproc.macros={compiler.path}{compiler.cpp.cmd} {compiler.cxx.flags} {build.extra_flags} {includes} -I{runtime.platform.path}/cores/{build.core} -I{build.path}/sketch -w -x c++ -E -CC "{source_file}" -o "{preprocessed_file_path}"

# Create static archive from objects
# Ensure archive exists at both {build.path}/{archive_file} and {build.path}/core/<basename>
# to be compatible with different CLI/tool behaviours that expect either path.
recipe.ar.pattern={compiler.path}{compiler.ar.cmd} {compiler.ar.flags} {compiler.ar.extra_flags} "{archive_file_path}" "{object_file}"

# Link objects into a native executable
# Link objects and core archive into a native executable.
# Use the core archive and all object files so the linker produces the final .out
# Link (C) and then run the produced binary.
# We wrap the link in sh -c so we can run the produced .out immediately after linking.
# Link (C) and then run the produced binary.
# We wrap the link in sh -c so we can run the produced .out immediately after linking.
recipe.c.link.pattern={compiler.path}{compiler.cpp.cmd} {compiler.ldflags} {object_files} {archive_file_path} {build.extra_flags} -o {build.path}/{build.project_name}.out
# Ensure both C and C++ link recipes exist. Some CLI versions call the C++ link recipe
# when linking C++ sketches, so provide an identical cpp variant.

# Link (C++) and then run the produced binary.
recipe.cpp.link.pattern={compiler.path}{compiler.cpp.cmd} {compiler.ldflags} {object_files} {archive_file_path} {build.extra_flags} -o {build.path}/{build.project_name}.out

# ELF/linker (combine) settings
# Some Arduino CLI/toolchains call a "combine" recipe to merge objects and archives
# into the final ELF; provide a combine pattern that uses --start-group/--end-group
# to ensure all libs are searched. We map compiler.c.elf.cmd to the C++ compiler by default.
compiler.c.elf.cmd={compiler.cpp.cmd}
compiler.c.elf.flags={compiler.cxx.flags}

# Combine (link) pattern: combine all object files and the core archive into the final executable.
# Using start-group/end-group helps resolve circular deps between archives and objects.
recipe.c.combine.pattern={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} -Wl,--start-group {object_files} {archive_file_path} -Wl,--end-group {build.extra_flags} -o {build.path}/{build.project_name}.out
recipe.cpp.combine.pattern={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} -Wl,--start-group {object_files} {archive_file_path} -Wl,--end-group {build.extra_flags} -o {build.path}/{build.project_name}.out
# macOS variant: Apple's ld64 does not support GNU "--start-group/--end-group"
# (and does not need them — it searches archives regardless of order).
recipe.c.combine.pattern.macosx={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} {object_files} {archive_file_path} {build.extra_flags} -o {build.path}/{build.project_name}.out
recipe.cpp.combine.pattern.macosx={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} {object_files} {archive_file_path} {build.extra_flags} -o {build.path}/{build.project_name}.out

# Upload recipe: execute the produced binary locally
tools.run.upload.pattern="{build.path}/{build.project_name}.out"
tools.run.upload.pattern.windows="cmd" /c start "" cmd /c "{build.path}/{build.project_name}.out"

# Size command
recipe.size.pattern={compiler.path}{compiler.size.cmd} {build.path}/{build.project_name}.out

# Core name
compiler.core=core
